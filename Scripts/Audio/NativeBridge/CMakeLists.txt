cmake_minimum_required(VERSION 3.20)
project(MHGZRiseAudioBridge LANGUAGES C CXX ASM_MASM)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This narrowly scoped Rise bridge requires MSVC x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(MHGZ_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(MINHOOK_DIR "${MHGZ_ROOT}/Saved/Audio/RiseIG/NativeBridge/ThirdParty/minhook" CACHE PATH "Pinned MinHook v1.3.4")
if(NOT EXISTS "${MINHOOK_DIR}/include/MinHook.h")
  message(FATAL_ERROR "Fetch the documented pinned MinHook into Saved first")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${MINHOOK_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/minhook" EXCLUDE_FROM_ALL)
add_executable(native_bridge_tests native_tests.cpp passthrough.asm trigger_fixture.asm)
target_link_libraries(native_bridge_tests PRIVATE minhook)
target_compile_options(native_bridge_tests PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
add_library(MHGZ_RiseAudioBridge SHARED plugin.cpp passthrough.asm)
target_link_libraries(MHGZ_RiseAudioBridge PRIVATE minhook bcrypt)
target_compile_definitions(MHGZ_RiseAudioBridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(MHGZ_RiseAudioBridge PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
enable_testing()
add_test(NAME NativeBridge.ScalarQueueAndRealDetour COMMAND native_bridge_tests)
add_executable(plugin_io_tests plugin_io_tests.cpp passthrough.asm)
target_link_libraries(plugin_io_tests PRIVATE minhook bcrypt)
target_compile_definitions(plugin_io_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(plugin_io_tests PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
add_test(NAME NativeBridge.ProductionIPCAndCSV COMMAND plugin_io_tests)
add_executable(plugin_abi_tests plugin_abi_tests.cpp)
add_dependencies(plugin_abi_tests MHGZ_RiseAudioBridge)
add_test(NAME NativeBridge.ActualDLLAndHostABI COMMAND plugin_abi_tests "$<TARGET_FILE:MHGZ_RiseAudioBridge>")
